# 元类型

元类型是一种特殊的 “字符串” 类型，其使用单引号 `''` 包裹，用于描述数据类型的属性，是一种“描述数据类型的数据类型”。

## 元类型基本定义

使用 `''` 定义元类型：

```losu
var print import print;

var meta = 'This is a Meta';

print(meta);
```

## 设置变量的元类型

通过设置 `Unit` 类型的 `‘type’` 对象可以设置其元类型，该方法常用于自定义数据中：

```losu
var print import print;

var point = {
  'type': 'Point',
  x: 1,
  y: 2
};

print(point->*Point)
print(point['type'])
```

## 使用元类型进行运算符重载

通过设置 `Unit` 类型中的同名元属性，可以对运算符进行重载：

```losu
var print import print

var PointMaker = {
  x, y =>
  // 保存构造函数
  var __init__ = _ ;
  // 返回一个 Unit
  return {
    'type': 'point',
    value: {
      x: x,
      y: y
    },
    // 重载 Unit 的 + 运算符
    '+': {
      // 执行 + 时会自动调用左值 '+' 对应的函数，并传入左右值作为 self、other
      self, other =>
      return  __init__(self.value.x + other.value.x, self.value.y + other.value.y)
    }
  }
};

var Point = PointMaker(1, 2) + PointMaker(2, 3)
print(Point.value.x->&String .. "  " .. Point.value.y->&String)
```

## 数据绑定元

通过增加数据绑定元（ `'set'`, `'get'` ）的成员函数，可以设置当某数据被修改时触发回调函数，例如：

```losu
var print import print

/* --------- set ---------- */

print("set 测试:\n")

// set 全局设置
var losu ={
  "text1": "",
  "text2": "",
  // 此时会自动传入 self,key,value
  'set': { self , k , v =>
      print(v)
  }
}
losu.text1 = "set text1" // 此时会输出 set text1
losu.text2 = "set text2"  // 此时会输出 set text2


// set 局部设置
var losu = {
  "text1": "",
  "text2": "",
  // 此时会自动传入 self,value
  'set': {
    text1:{
      self, v =>
        print(v)
    }
  }
}
losu.text1 = "set text1" // 此时会输出 set text1
losu.text2 = "set text2" // 不会有输出

// --------- get ----------

print("\nget 测试:\n")

// get 全局设置
var losu ={
  "text1": "",
  "text2": "",
  // 此时会自动传入 self,key
  'get': { self, k =>
      print("get " .. k)
  }
}
losu.text1 // 此时会输出 text1
losu.text2 // 此时会输出 text2


// get 局部设置
var losu = {
  "text1": "",
  "text2": "",
  // self
  'get': {
    text1:{
      self =>
        print("get text1")
    }
  }
}
losu.text1 // 此时会输出 get text1
losu.text2 // 不会有输出
```

## 练习

### 练习 1：使用元类型实现向量类

利用元类型的运算符重载功能，实现一个简单的二维向量（Vector2D），支持加法运算和类型检查。

<details>
<summary>点击查看答案</summary>

```losu
var print import print;

// 定义向量构造函数
var Vector2D = {
  x, y =>
  var __init__ = _ ;
  return {
    'type': 'Vector2D',
    x: x,
    y: y,
    '+': {
      self, other =>
      if (other->*Vector2D == nil) {
        print("错误：只能与 Vector2D 类型相加")
        return nil
      }
      return __init__(self.x + other.x, self.y + other.y)
    }
  }
};

// 创建向量
var v1 = Vector2D(3, 4)
var v2 = Vector2D(1, 2)

print("=== 向量加法测试 ===")
print("v1: (" .. v1.x->&String .. ", " .. v1.y->&String .. ")")
print("v2: (" .. v2.x->&String .. ", " .. v2.y->&String .. ")")

var v3 = v1 + v2
print("v1 + v2: (" .. v3.x->&String .. ", " .. v3.y->&String .. ")")

print("\n=== 类型检查 ===")
print("v1->*Vector2D: " .. (v1->*Vector2D != nil)->&String)
print("v3->*Vector2D: " .. (v3->*Vector2D != nil)->&String)
```

</details>

### 练习 2：使用数据绑定元实现响应式对象

利用 `'set'` 和 `'get'` 数据绑定元，创建一个具有自动日志功能的配置对象，能够记录属性的读写操作。

<details>
<summary>点击查看答案</summary>

```losu
var print import print;

// 创建带日志的配置对象
var config = {
  host: "localhost",
  port: 8080,
  debug: false,

  // 全局 set 回调
  'set': {
    self, key, value =>
    print("[SET] " .. key .. " = " .. value->&String)
  },

  // 局部 get 回调（仅针对 port）
  'get': {
    port: {
      self =>
      print("[GET] 访问 port 属性")
    }
  }
}

print("=== 修改属性测试 ===")
config.host = "192.168.1.1"
config.port = 3000
config.debug = true

print("\n=== 读取属性测试 ===")
print("当前 host: " .. config.host)
print("当前 port: " .. config.port->&String)
print("当前 debug: " .. config.debug->&String)

print("\n=== 再次读取 port（触发 get 回调）===")
var p = config.port
print("获取到的端口值：" .. p->&String)
```

</details>

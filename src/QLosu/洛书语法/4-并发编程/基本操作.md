# 基本操作

洛书线程提供了一系列基本操作来控制线程的执行流程，包括创建、让出、取消、等待、休眠等。这些操作使得开发者能够精确控制并发程序的行为。

> [!WARNING]
> 要使用多线程，必须将程序写入 `main{}` 块中，并使用 `async` 函数创建线程。

## 创建线程（async）

`async` 函数用于创建新的洛书线程，其原型是std库中的`std::async`函数：

```losu
import std

main {
    println('=== 线程创建演示 ===')

    # 创建线程并立即执行
    async(true) {
        println('新线程: 我被创建了')
    }

    # 创建线程并获取线程对象
    let thread = async(true) {
        println('新线程: 我有线程对象')
    }

    println('主线程: 线程对象', thread)
}
```

## 让出控制权（yield）

`yield` 函数让当前线程主动让出CPU控制权：

```losu
import std

main {
    println('=== 线程让出演示 ===')

    async(true) {
        println('线程1: 开始执行')
        yield  # 让出控制权
        println('线程1: 恢复执行')
    }

    async(true) {
        println('线程2: 开始执行')
        yield  # 让出控制权
        println('线程2: 恢复执行')
    }

    println('主线程: 创建完成')
    yield  # 主线程也让出控制权
    println('主线程: 恢复执行')
}
```

## 等待线程（await）

`await` 函数等待指定线程执行完毕：

```losu
import std

main {
    println('=== 线程等待演示 ===')

    let worker = async(true) {
        for (i in 1, 3) {
            println('工作线程: 步骤', i)
            yield
        }
        println('工作线程: 任务完成')
    }

    println('主线程: 等待工作线程完成')
    let result = await(worker)
    println('主线程: 工作线程已完成，结果:', result)
}
```

当设置 `await` 函数的第二个参数为 `true` 时，`await` 会进入 peek 模式，即判断当前的线程是否执行完毕。若线程执行完毕时，`await` 会返回线程的返回值 unit，否则返回 false。

```losu
import std

main {
    println('=== 线程等待演示 ===')

    let worker = async(true) {
        for (i in 1, 3) {
            println('工作线程: 步骤', i)
            yield
        }
        println('工作线程: 任务完成')
    }

    println('主线程: 等待工作线程完成')
    let result = await(worker, true)
    println('主线程: 对线程进行 peek ，结果:', result)
    sleep(1000)
    let result2 = await(worker, true)
    println('主线程: 工作线程已完成，结果:', result2)
}
```

## 取消线程（cancel）

`cancel` 函数取消指定的线程：

```losu
import std

main {
    println('=== 线程取消演示 ===')

    let longTask = async(true) {
        for (i in 1, 10) {
            println('长任务: 步骤', i)
            yield
        }
        println('长任务: 全部完成')
    }

    # 让长任务执行几步
    for (i in 1, 3) {
        println('主线程: 让长任务执行')
        yield
    }

    # 取消长任务
    println('主线程: 取消长任务')
    cancel(longTask)

    println('主线程: 任务已取消')
}
```

## 休眠（sleep）

`sleep` 函数让当前线程休眠指定的毫秒数：

```losu
import std

main {
    println('=== 线程休眠演示 ===')

    async(true) {
        for (i in 1, 3) {
            println('线程1: 计数', i)
            sleep(100)  # 休眠100毫秒
            yield
        }
        println('线程1: 完成')
    }

    async(true) {
        for (i in 1, 3) {
            println('线程2: 计数', i)
            sleep(200)  # 休眠200毫秒
            yield
        }
        println('线程2: 完成')
    }

    println('主线程: 等待所有线程完成')
}
```

## 同步机制

洛书内置了自动同步机制，多个线程可以安全地访问共享变量：

```losu
import std

main {
    println('=== 同步机制演示 ===')

    global counter = 0

    let thread1 = async(true) {
        for (i in 1, 5) {
            counter = counter + 1
            println('线程1: counter =', counter)
            yield
        }
    }

    let thread2 = async(true) {
        for (i in 1, 5) {
            counter = counter + 1
            println('线程2: counter =', counter)
            yield
        }
    }

    await(thread1)
    await(thread2)

    println('最终结果: counter =', counter)
}
```

## 复杂示例

### 生产者-消费者模式

```losu
import std

main {
    println('=== 生产者-消费者演示 ===')

    global buffer = []
    global maxSize = 3

    # 生产者线程
    let producer = async(true) {
        for (i in 1, 5) {
            if (sizeof(buffer) < maxSize) {
                buffer = buffer + [i]
                println('生产者: 生产了', i, '缓冲区:', buffer)
            } else {
                println('生产者: 缓冲区满了，等待')
            }
            yield
        }
        println('生产者: 完成')
    }

    # 消费者线程
    let consumer = async(true) {
        for (i in 1, 5) {
            if (sizeof(buffer) > 0) {
                let item = buffer[0]
                # 简单移除第一个元素（实际实现可能需要更复杂的数组操作）
                println('消费者: 消费了', item, '缓冲区大小:', sizeof(buffer))
            } else {
                println('消费者: 缓冲区空了，等待')
            }
            yield
        }
        println('消费者: 完成')
    }

    await(producer)
    await(consumer)
    println('演示完成')
}
```

### 练习 2：任务调度

创建一个简单的任务调度系统，演示线程的创建、等待和取消。

<details>
<summary>查看答案</summary>

```losu
import std

main {
    println('=== 任务调度演示 ===')

    # 创建短任务
    let shortTask = async(true) {
        for (i in 1, 2) {
            println('短任务: 步骤', i)
            yield
        }
        println('短任务: 完成')
    }

    # 创建长任务
    let longTask = async(true) {
        for (i in 1, 6) {
            println('长任务: 步骤', i)
            yield
        }
        println('长任务: 完成')
    }

    # 等待短任务完成
    println('调度器: 等待短任务')
    await(shortTask)

    # 让长任务执行一会儿
    println('调度器: 让长任务执行')
    yield
    yield

    # 取消长任务
    println('调度器: 取消长任务')
    cancel(longTask)

    println('调度器: 任务调度完成')
}
```

</details>
